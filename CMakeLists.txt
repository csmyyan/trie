cmake_minimum_required(VERSION 3.20)
project(cow_trie_lab LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(trie_core src/trie.cpp src/trie_store.cpp src/epoch.cpp)
target_include_directories(trie_core PUBLIC include ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(trie_core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(trie_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

add_executable(trie_bench test/trie_bench.cpp)
target_link_libraries(trie_bench PRIVATE trie_core)

include(CTest)
if(BUILD_TESTING)
  foreach(name trie trie_store epoch trie_performance)
    add_executable(${name}_test test/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE trie_core)
  endforeach()

  add_test(NAME provided COMMAND trie_test StarterSupport.)
  add_test(NAME trie COMMAND trie_test TrieTest.)
  add_test(NAME trie_store COMMAND trie_store_test TrieStoreTest.)
  add_test(NAME cas_trie_store COMMAND trie_store_test CasStoreTest.)
  add_test(NAME epoch_provided COMMAND epoch_test StarterSupport.)
  add_test(NAME epoch COMMAND epoch_test EpochTest.)
  add_test(NAME epoch_trie_store COMMAND epoch_test EpochTrieStoreTest.)
  add_test(NAME benchmark_smoke COMMAND trie_bench 100 2 32)
  set_tests_properties(provided epoch_provided PROPERTIES LABELS provided TIMEOUT 10)
  set_tests_properties(trie PROPERTIES LABELS task1 TIMEOUT 120)
  set_tests_properties(trie_store PROPERTIES LABELS "task2;mutex" TIMEOUT 120)
  set_tests_properties(cas_trie_store PROPERTIES LABELS "task2;cas" TIMEOUT 120)
  set_tests_properties(epoch PROPERTIES LABELS "task3;epoch" TIMEOUT 30)
  set_tests_properties(epoch_trie_store PROPERTIES LABELS "task3;epoch_store" TIMEOUT 60)
  set_tests_properties(benchmark_smoke PROPERTIES LABELS benchmark TIMEOUT 30)

  add_test(NAME trie_performance COMMAND trie_performance_test TriePerformance.)
  add_test(NAME trie_store_performance COMMAND trie_performance_test StorePerformance.)
  add_test(NAME cas_trie_store_performance COMMAND trie_performance_test CasPerformance.)
  set_tests_properties(trie_performance PROPERTIES LABELS "task1;performance")
  set_tests_properties(trie_store_performance PROPERTIES LABELS "task2;mutex;performance")
  set_tests_properties(cas_trie_store_performance PROPERTIES LABELS "task2;cas;performance")
  set_tests_properties(trie_performance trie_store_performance cas_trie_store_performance PROPERTIES
    RUN_SERIAL TRUE TIMEOUT 240)
endif()
